blinky_pcb

A circuit board: an LED blinker powered from USB-C. A 555 timer in astable mode flashes the LED about once a second (R1 10k, R2 47k, C1 10u), the two 5.1k resistors on CC1/CC2 ask the USB-C source for 5 V, and C3 decouples the timer. The design is written as a native KiCad project (schematic, board, project) with parts from KiCad's own libraries, routed by hand below, the pours filled by KiCad. Then KiCad's ERC and DRC check it, the fabrication files are written, and the board's STEP opens as a cad model. Needs KiCad 9 or newer installed (its kicad-cli and libraries). Run: cad run examples/blinky_pcb.ts (writes out/blinky/, checks it, shows the board) cad run examples/blinky_pcb.ts --fab (also writes out/blinky-fab/: gerbers, drill, BOM...)

// A circuit board: an LED blinker powered from USB-C. A 555 timer in astable mode flashes the
// LED about once a second (R1 10k, R2 47k, C1 10u), the two 5.1k resistors on CC1/CC2 ask the
// USB-C source for 5 V, and C3 decouples the timer.
//
// The design is written as a native KiCad project (schematic, board, project) with parts from
// KiCad's own libraries, routed by hand below, the pours filled by KiCad. Then KiCad's ERC and DRC
// check it, the fabrication files are written, and the board's STEP opens as a cad model.
// Needs KiCad 9 or newer installed (its kicad-cli and libraries).
//
// Run: cad run examples/blinky_pcb.ts           (writes out/blinky/, checks it, shows the board)
//      cad run examples/blinky_pcb.ts --fab     (also writes out/blinky-fab/: gerbers, drill, BOM...)
import { importStep, model, pcb, rect } from "@cad/sdk";

export function blinky(): pcb.PcbDesign {
  const d = pcb.design("blinky", { title: "USB-C LED blinker" });
  const R = "Resistor_SMD:R_0805_2012Metric", C = "Capacitor_SMD:C_0805_2012Metric";

  // Parts, placed in mm with y up from the board's centre; rotation in degrees.
  const j1 = d.part("J1", { symbol: "Connector:USB_C_Receptacle_PowerOnly_6P", footprint: "Connector_USB:USB_C_Receptacle_GCT_USB4125-xx-x_6P_TopMnt_Horizontal", value: "USB-C", at: [0, -8.5] });
  const u1 = d.part("U1", { symbol: "Timer:NE555D", footprint: "Package_SO:SOIC-8_3.9x4.9mm_P1.27mm", at: [0, 2] });
  const r1 = d.part("R1", { symbol: "Device:R", footprint: R, value: "10k", at: [6.5, 6.5], rotation: 270 });
  const r2 = d.part("R2", { symbol: "Device:R", footprint: R, value: "47k", at: [6.5, 3], rotation: 270 });
  const c1 = d.part("C1", { symbol: "Device:C", footprint: C, value: "10u", at: [8.2, -1.5], rotation: 270 });
  const c2 = d.part("C2", { symbol: "Device:C", footprint: C, value: "10n", at: [5.2, -1.8] });
  const c3 = d.part("C3", { symbol: "Device:C", footprint: C, value: "100n", at: [0, 6.5], rotation: 180 });
  const r3 = d.part("R3", { symbol: "Device:R", footprint: R, value: "330", at: [-7, 1.365], rotation: 180 });
  const d1 = d.part("D1", { symbol: "Device:LED", footprint: "LED_SMD:LED_0805_2012Metric", value: "red", at: [-11, 1.365] });
  const r4 = d.part("R4", { symbol: "Device:R", footprint: R, value: "5.1k", at: [-10, -6], rotation: 180 });
  const r5 = d.part("R5", { symbol: "Device:R", footprint: R, value: "5.1k", at: [10, -6] });

  // Nets: pins are the symbol's pin numbers (the footprint's pads).
  d.net("VBUS", j1.pin("A9"), j1.pin("B9"), u1.pin(8), u1.pin(4), r1.pin(1), c3.pin(1));
  d.net("GND", j1.pin("A12"), j1.pin("B12"), j1.pin("SH"), u1.pin(1), c1.pin(2), c2.pin(2), d1.pin(1), c3.pin(2), r4.pin(2), r5.pin(2));
  d.net("CC1", j1.pin("A5"), r4.pin(1));
  d.net("CC2", j1.pin("B5"), r5.pin(1));
  d.net("DIS", u1.pin(7), r1.pin(2), r2.pin(1));
  d.net("THR", u1.pin(6), u1.pin(2), r2.pin(2), c1.pin(1));
  d.net("CTRL", u1.pin(5), c2.pin(1));
  d.net("OUT", u1.pin(3), r3.pin(1));
  d.net("LED_A", r3.pin(2), d1.pin(2));

  // The board: a 40 × 25 mm outline with 2 mm corners, drawn as any cad profile.
  d.outline(rect(40, 25, { radius: 2 }));

  // Routing: signals on top, crossings and the VBUS trunk on the bottom, GND poured on both.
  const top = (net: string, pts: [number, number][], width = 0.25) => d.track(net, pts, { width });
  const bottom = (net: string, pts: [number, number][], width = 0.25) => d.track(net, pts, { width, layer: "B.Cu" });
  top("DIS", [[2.475, 2.635], [3.5, 2.635], [4.777, 3.912], [6.5, 3.912]]);
  top("DIS", [[6.5, 5.588], [6.5, 3.912]]);
  top("THR", [[6.5, 2.088], [6.5, 0.5], [8.2, -0.588]]);
  top("THR", [[2.475, 1.365], [6.5, 1.365]]);
  top("THR", [[-2.475, 2.635], [-4.3, 2.635]]);
  bottom("THR", [[-4.3, 2.635], [4.3, 1.365]]);
  top("CTRL", [[2.475, 0.095], [3.2, 0.095], [4.288, -1.0], [4.288, -1.8]]);
  top("VBUS", [[2.475, 3.905], [3.5, 3.905], [3.5, 7.412], [6.5, 7.412], [9, 7.412]], 0.4);
  top("VBUS", [[0.95, 6.5], [3.5, 6.5]], 0.4);
  top("GND", [[-0.95, 6.5], [-3.5, 6.5], [-3.5, 3.905], [-2.475, 3.905]], 0.4);
  top("VBUS", [[1.52, -5.42], [1.52, -3.0]], 0.4);
  top("VBUS", [[-1.52, -5.42], [-1.52, -3.0]], 0.4);
  bottom("VBUS", [[-4.5, -3.0], [-1.52, -3.0]], 0.4);
  bottom("VBUS", [[-1.52, -3.0], [1.52, -3.0], [9, -3.0], [9, 7.412]], 0.4);
  bottom("VBUS", [[-4.5, -3.0], [-4.5, 0.095]], 0.4);
  top("VBUS", [[-4.5, 0.095], [-2.475, 0.095]], 0.4);
  top("OUT", [[-2.475, 1.365], [-6.088, 1.365]]);
  top("LED_A", [[-7.912, 1.365], [-10.062, 1.365]]);
  top("CC1", [[-0.5, -5.42], [-0.5, -4.3]]);
  bottom("CC1", [[-0.5, -4.3], [-7.5, -4.3]]);
  top("CC1", [[-7.5, -4.3], [-7.5, -6], [-9.088, -6]]);
  top("CC2", [[0.5, -5.42], [0.5, -4.3]]);
  bottom("CC2", [[0.5, -4.3], [7.5, -4.3]]);
  top("CC2", [[7.5, -4.3], [7.5, -6], [9.088, -6]]);
  for (const [net, at] of [
    ["THR", [-4.3, 2.635]], ["THR", [4.3, 1.365]], ["VBUS", [9, 7.412]], ["VBUS", [1.52, -3.0]], ["VBUS", [-1.52, -3.0]],
    ["VBUS", [-4.5, 0.095]], ["CC1", [-0.5, -4.3]], ["CC1", [-7.5, -4.3]], ["CC2", [0.5, -4.3]], ["CC2", [7.5, -4.3]],
    ["GND", [-13, -9]], ["GND", [13, -9]], ["GND", [-13, 9]], ["GND", [13, 9]], ["GND", [-3, 9.5]],
  ] as [string, [number, number]][]) d.via(net, at);
  d.zone("GND", { layer: "F.Cu" }).zone("GND", { layer: "B.Cu" });
  return d;
}

if (import.meta.main) {
  const files = await blinky().write("out/blinky", { fill: true });
  for (const n of files.notes) console.log("note:", n);
  const erc = await pcb.erc(files.schematic);
  const drc = await pcb.drc(files.board, { parity: true });
  for (const r of [erc, drc]) {
    console.log(`${r.check.toUpperCase()} ${r.ok ? "passed" : "FAILED"}: ${r.errors} errors, ${r.warnings} warnings`);
    for (const f of r.findings.filter((f) => f.severity === "error")) console.log(`  ${f.rule}: ${f.message} [${f.refs.join(", ")}]`);
  }
  if (process.argv.includes("--fab")) {
    for (const f of await pcb.fabricate(files.board, ["gerbers", "drill", "pos", "bom", "pdf"], { out: "out/blinky-fab" })) console.log("wrote", f);
  }
  // The board with its parts' 3D models, as a cad model in the viewer (and for an enclosure).
  const stepFile = await pcb.step(files.board, { out: "out/blinky/blinky.step" });
  await model("blinky_board", () => importStep(stepFile))();
  if (!erc.ok || !drc.ok) process.exitCode = 1;
}

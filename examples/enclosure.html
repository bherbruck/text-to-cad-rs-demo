// A two-part enclosure: a shelled base with screw bosses and a lid with a locating lip, each its
// own model (to print or export alone) and together as an assembly.
// Run: cad run examples/enclosure.ts
import { assembly, box, cylinder, model, param } from "@cad/sdk";

const W = 90, D = 60, H = 32;         // outside size of the base
const LID = 3;                         // lid plate thickness
const R = 5;                           // outside corner radius
const BOSS_INSET = 7;                  // boss centre from the outside walls

function base(wall: number) {
  const outer = box(W, D, H, { align: ["center", "center", "min"] }).fillet("|Z", R);
  const shell = outer.shell(">Z", -wall);                         // open top, walls grow inward
  const boss = cylinder(3.5, H - wall, { align: ["center", "center", "min"] }).translate(0, 0, wall);
  const pilot = cylinder(1.25, H, { align: ["center", "center", "min"] }).translate(0, 0, wall + 2);
  const sx = W - 2 * BOSS_INSET, sy = D - 2 * BOSS_INSET;
  return shell
    .fuse(boss.gridCopies(2, 2, sx, sy))
    .cut(pilot.gridCopies(2, 2, sx, sy));
}

function lid(wall: number, clearance: number) {
  const plate = box(W, D, LID, { align: ["center", "center", "min"] }).fillet("|Z", R);
  const lipW = W - 2 * wall - 2 * clearance, lipD = D - 2 * wall - 2 * clearance;
  const lip = box(lipW, lipD, 4, { align: ["center", "center", "max"] })
    .fillet("|Z", R - wall)
    .shell("<Z", -1.6);
  const screw = cylinder(1.7, 20).gridCopies(2, 2, W - 2 * BOSS_INSET, D - 2 * BOSS_INSET);
  // the lip stops short of each corner's boss: a square notch, so no sliver of lip is left
  // between the boss's clearance and the wall (a 0.6 mm sliver fails a printing check)
  const notch = 2 * (BOSS_INSET + 4.5);
  const bossClear = box(notch, notch, 10, { align: ["center", "center", "max"] }).gridCopies(2, 2, W - 2 * BOSS_INSET, D - 2 * BOSS_INSET);
  return plate.fuse(lip).cut(screw, bossClear).chamfer(">Z", 0.8);
}

const wallParam = () => param("wall", 2, { min: 1.2, max: 4, step: 0.2, unit: "mm" });

export const enclosureBase = model("enclosure_base", () =>
  base(wallParam()).label("base", { color: "#6b8aa8", material: "petg" }));

export const enclosureLid = model("enclosure_lid", () =>
  lid(wallParam(), param("clearance", 0.2, { min: 0, max: 1, step: 0.05, unit: "mm" }))
    .label("lid", { color: "#d8dde3", material: "petg" }));

export const enclosure = model("enclosure", () => {
  const wall = wallParam();
  const open = param("lid_lift", 15, { min: 0, max: 60, unit: "mm", description: "exploded gap" });
  return assembly("enclosure", [
    base(wall).label("base", { color: "#6b8aa8", material: "petg" }),
    lid(wall, 0.2).label("lid", { color: "#d8dde3", material: "petg" }).translate(0, 0, H + open),
  ]);
});

if (import.meta.main) {
  await enclosureBase();
  await enclosureLid();
  await enclosure();
}

01-nema17-mount

A parametric NEMA 17 motor mount: an L bracket whose upright takes the motor (22 mm pilot, four M3 slots on the 31 mm square so the belt can be tensioned) and whose base bolts down through two M5 slots. Check: cad model dfm "$ID" --process cnc --material aluminum --tools 3,6

#Source

Run it with cad run examples/dogfood/01-nema17-mount.ts.

// A parametric NEMA 17 motor mount: an L bracket whose upright takes the motor (22 mm pilot,
// four M3 slots on the 31 mm square so the belt can be tensioned) and whose base bolts down
// through two M5 slots. Check: cad model dfm "$ID" --process cnc --material aluminum --tools 3,6
import { Plane, box, model, param, sel, slot } from "@cad/sdk";

export const nemaMount = model("nema17_mount", () => {
  const t = param("thickness", 4, { min: 3, max: 8, step: 0.5, unit: "mm" });
  const travel = param("slot_travel", 4, { min: 0, max: 10, step: 0.5, unit: "mm", description: "belt tension adjustment" });
  const W = 50, H = 50, D = 40;

  // upright in the XZ plane at y = 0..t; base along +Y
  const upright = box(W, t, H, { align: ["center", "min", "min"] });
  const base = box(W, D, t, { align: ["center", "min", "min"] });
  let body = upright.fuse(base).fillet(sel.near([0, t, t]), 3);

  // the motor's face: pilot and four M3 slots on a 31 mm square, centred at z = 27
  const cz = 27;
  const onFace = Plane.XZ.moveTo([0, t + 1, cz]); // XZ's normal is -Y: extrude toward -Y through the plate
  const pilot = slot(22 + travel, 22, { plane: onFace }).extrude(t + 2);
  const m3 = [[-15.5, -15.5], [15.5, -15.5], [15.5, 15.5], [-15.5, 15.5]].map(([u, v]) =>
    slot(3.4 + travel, 3.4, { plane: onFace }).at(u!, v!).extrude(t + 2));
  // base slots for M5 along Y
  const onBase = Plane.XY.moveTo([0, 0, -1]);
  const m5 = [-17, 17].map((u) => slot(5.5 + 8, 5.5, { plane: onBase }).turn(90).at(u, 26).extrude(t + 2));

  body = body.cut(pilot, m3, m5).fillet("|Z and (<X or >X) and >Y", 3);
  return body.label("mount", { color: "#4a6fa5", material: "aluminum" });
});

if (import.meta.main) await nemaMount();

esp32_pcb

A denser board, autorouted: an ESP32-C3 breakout on four layers. USB-C into a CH340C USB-serial bridge and an AMS1117 3.3 V regulator, EN and BOOT buttons, a status LED, and two 8-pin headers. GND and 3V3 are inner planes; the router fans each of their pads out to a via, and routes every other net on the outer layers. Needs KiCad 9 or newer installed. Run: cad run examples/esp32_pcb.ts (writes out/esp32c3/, routes, checks and shows it)

#Source

Run it with cad run examples/esp32_pcb.ts.

// A denser board, autorouted: an ESP32-C3 breakout on four layers. USB-C into a CH340C
// USB-serial bridge and an AMS1117 3.3 V regulator, EN and BOOT buttons, a status LED, and two
// 8-pin headers. GND and 3V3 are inner planes; the router fans each of their pads out to a via,
// and routes every other net on the outer layers.
// Needs KiCad 9 or newer installed.
//
// Run: cad run examples/esp32_pcb.ts     (writes out/esp32c3/, routes, checks and shows it)
import { importStep, model, pcb, rect } from "@cad/sdk";

export function esp32(): pcb.PcbDesign {
  const d = pcb.design("esp32c3", { title: "ESP32-C3 breakout" });
  const R = "Resistor_SMD:R_0603_1608Metric", C = "Capacitor_SMD:C_0603_1608Metric";
  const part = (ref: string, symbol: string, footprint: string, value: string, at: [number, number], rotation = 0) =>
    d.part(ref, { symbol, footprint, value, at, rotation });

  part("U1", "RF_Module:ESP32-C3-WROOM-02", "RF_Module:ESP32-C3-WROOM-02", "ESP32-C3-WROOM-02", [14, 8]);
  part("U2", "Interface_USB:CH340C", "Package_SO:SOIC-16_3.9x9.9mm_P1.27mm", "CH340C", [-10, 0]);
  part("U3", "Regulator_Linear:AMS1117-3.3", "Package_TO_SOT_SMD:SOT-223-3_TabPin2", "AMS1117-3.3", [15, -10]);
  part("J1", "Connector:USB_C_Receptacle_USB2.0_16P", "Connector_USB:USB_C_Receptacle_GCT_USB4105-xx-A_16P_TopMnt_Horizontal", "USB-C", [0, -15.5]);
  part("J2", "Connector_Generic:Conn_01x08", "Connector_PinHeader_2.54mm:PinHeader_1x08_P2.54mm_Vertical", "IO-L", [-27, 8.89]);
  part("J3", "Connector_Generic:Conn_01x08", "Connector_PinHeader_2.54mm:PinHeader_1x08_P2.54mm_Vertical", "IO-R", [27, 8.89]);
  part("C1", "Device:C", C, "10u", [22, -14], 90);
  part("C2", "Device:C", C, "10u", [22, -8], 90);
  part("C3", "Device:C", C, "100n", [-10, -7.5]);
  part("C4", "Device:C", C, "100n", [-10, 7.5]);
  part("C5", "Device:C", C, "100n", [3, 10], 90);
  part("C6", "Device:C", C, "1u", [-4, 12], 90);
  part("R1", "Device:R", R, "5.1k", [-8, -13], 90);
  part("R2", "Device:R", R, "5.1k", [8, -13], 90);
  part("R3", "Device:R", R, "10k", [-4, 16], 90);
  part("R4", "Device:R", R, "10k", [-8, 12], 90);
  part("R5", "Device:R", R, "1k", [2, -6]);
  part("D1", "Device:LED", "LED_SMD:LED_0603_1608Metric", "green", [6, -6]);
  part("SW1", "Switch:SW_Push", "Button_Switch_SMD:SW_SPST_PTS810", "EN", [-12, 15]);
  part("SW2", "Switch:SW_Push", "Button_Switch_SMD:SW_SPST_PTS810", "BOOT", [-20, 15]);

  const nets: Record<string, string[]> = {
    VBUS: ["J1.A4", "J1.A9", "J1.B4", "J1.B9", "U3.3", "C1.1", "J3.7", "U2.16", "C4.1"],
    V3: ["U2.4", "C3.1"],
    GND: ["J1.A1", "J1.A12", "J1.B1", "J1.B12", "J1.SH", "U3.1", "C1.2", "C2.2", "C3.2", "C4.2", "C5.2", "C6.2", "U2.1", "U1.9", "U1.19", "R1.2", "R2.2", "SW1.2", "SW2.2", "J2.8", "J3.8", "D1.1"],
    "3V3": ["U3.2", "C2.1", "U1.1", "C5.1", "R3.1", "R4.1", "J2.1", "J3.1"],
    CC1: ["J1.A5", "R1.1"], CC2: ["J1.B5", "R2.1"],
    USB_DP: ["J1.A6", "J1.B6", "U2.5"], USB_DM: ["J1.A7", "J1.B7", "U2.6"],
    TXD: ["U2.2", "U1.11"], RXD: ["U2.3", "U1.12"],
    EN: ["U1.2", "R3.2", "C6.1", "SW1.1"], BOOT: ["U1.8", "R4.2", "SW2.1"],
    LED: ["U1.7", "R5.1"], LED_A: ["R5.2", "D1.2"],
    IO4: ["U1.3", "J2.2"], IO5: ["U1.4", "J2.3"], IO6: ["U1.5", "J2.4"], IO7: ["U1.6", "J2.5"], IO10: ["U1.10", "J2.6"], IO0: ["U1.18", "J2.7"],
    IO1: ["U1.17", "J3.2"], IO2: ["U1.16", "J3.3"], IO3: ["U1.15", "J3.4"], IO18: ["U1.13", "J3.5"], IO19: ["U1.14", "J3.6"],
  };
  for (const [name, pins] of Object.entries(nets)) d.net(name, ...pins);

  d.outline(rect(60, 40, { radius: 2 }));
  d.stackup({ layers: 4 });
  d.zone("GND", { layer: "In1.Cu" }).zone("3V3", { layer: "In2.Cu" });
  d.autoroute();
  return d;
}

if (import.meta.main) {
  const t = performance.now();
  const files = await esp32().write("out/esp32c3", { fill: true });
  console.log(`written and routed in ${((performance.now() - t) / 1000).toFixed(1)} s`);
  for (const n of files.notes) console.log("note:", n);
  const erc = await pcb.erc(files.schematic);
  const drc = await pcb.drc(files.board, { parity: true });
  for (const r of [erc, drc]) console.log(`${r.check.toUpperCase()} ${r.ok ? "passed" : "FAILED"}: ${r.errors} errors, ${r.warnings} warnings`);
  const stepFile = await pcb.step(files.board, { out: "out/esp32c3/esp32c3.step" });
  await model("esp32c3_board", () => importStep(stepFile))();
  if (!erc.ok || !drc.ok) process.exitCode = 1;
}

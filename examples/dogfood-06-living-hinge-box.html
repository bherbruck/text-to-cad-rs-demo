06-living-hinge-box

A clamshell box printed in one piece, open, without supports: two shallow trays side by side joined at their rims by a 0.5 mm living hinge (a 2 mm bridge), folding 180 degrees closed. A cantilever hook on the lid snaps into a window in the base's far wall; its barb's underside slopes at 45 degrees so it prints unsupported. Check: cad model dfm "$ID" --process printing --material pp

// A clamshell box printed in one piece, open, without supports: two shallow trays side by side
// joined at their rims by a 0.5 mm living hinge (a 2 mm bridge), folding 180 degrees closed. A
// cantilever hook on the lid snaps into a window in the base's far wall; its barb's underside
// slopes at 45 degrees so it prints unsupported.
// Check: cad model dfm "$ID" --process printing --material pp
import { assembly, box, model, param, polygon, revolute, Plane } from "@cad/sdk";

const L = 60, W = 40, wall = 2, gap = 2;

function tray(h: number) {
  return box(L, W, h, { align: ["center", "min", "min"] }).fillet("|Z", 3).shell(">Z", -wall);
}

export const clamshell = model("living_hinge_box", () => {
  const h = param("tray_height", 12, { min: 8, max: 30, unit: "mm" });
  const hinge = param("hinge", 0.5, { min: 0.3, max: 1, step: 0.05, unit: "mm", description: "living hinge thickness" });

  // base: y 0..W; the hinge strip spans the gap at the rims; a window for the barb in the far wall
  // it runs into the base's wall and stops at the lid's (touching, not overlapping: two parts here)
  const strip = box(L - 16, gap + 1, hinge, { align: ["center", "max", "max"] }).translate(0, 1, h);
  const window = box(12, wall + 2, 3, { align: ["center", "min", "min"] }).translate(0, W - wall - 1, h - 6);
  const base = tray(h).fuse(strip).cut(window);

  // lid: y -gap-W .. -gap, built open; its hook rises from the inside of its far wall
  const lidTray = tray(h).translate(0, -gap - W, 0);
  const hookY = -gap - W + wall;                                  // the far wall's inside face
  const hook = box(10, 1.6, 6 + 1, { align: ["center", "min", "min"] }).translate(0, hookY, h - 1);
  const barb = polygon([[hookY, h + 3.5], [hookY, h + 5.5], [hookY - 1, h + 5.5], [hookY - 1, h + 4.5]], { plane: Plane.YZ.moveTo([-5, 0, 0]) })
    .extrude(10);
  const lid = lidTray.fuse(hook, barb);

  const axis = { origin: [0, -gap / 2, h] as [number, number, number], dir: "-X" as const };   // right-handed about -X: the lid swings up and over
  return assembly("clamshell", [base.label("base", { color: "#4fd1c5", material: "pp" }), lid.label("lid", { color: "#81e6d9", material: "pp" })], {
    joints: [revolute("hinge", { parent: "base", child: "lid", axis, limits: [0, 180] })],
    poses: { open: { hinge: 0 }, closed: { hinge: 180 } },
  });
});

if (import.meta.main) await clamshell();

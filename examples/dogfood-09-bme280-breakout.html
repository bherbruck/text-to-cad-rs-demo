09-bme280-breakout

A BME280 sensor breakout: the sensor in I2C mode (CSB high, SDO low: address 0x76), 4.7k pull-ups on SDA and SCL, 100n decoupling, a 4-pin 2.54 mm header (3V3, GND, SDA, SCL). 20 x 16 mm, two layers: signals and 3V3 on top, SDA and SCL dropping through vias to reach the header on the bottom, GND poured on the bottom and stitched up to the sensor's ground pads. Needs KiCad 9 or newer installed. Run: cad run examples/dogfood/09-bme280-breakout.ts (writes out/bme280/, checks it, shows it)

// A BME280 sensor breakout: the sensor in I2C mode (CSB high, SDO low: address 0x76), 4.7k pull-ups
// on SDA and SCL, 100n decoupling, a 4-pin 2.54 mm header (3V3, GND, SDA, SCL). 20 x 16 mm, two
// layers: signals and 3V3 on top, SDA and SCL dropping through vias to reach the header on the
// bottom, GND poured on the bottom and stitched up to the sensor's ground pads.
// Needs KiCad 9 or newer installed.
// Run: cad run examples/dogfood/09-bme280-breakout.ts   (writes out/bme280/, checks it, shows it)
import { importStep, model, pcb, rect } from "@cad/sdk";

const d = pcb.design("bme280_breakout", { title: "BME280 breakout" });
const u1 = d.part("U1", { symbol: "Sensor:BME280", footprint: "Package_LGA:Bosch_LGA-8_2.5x2.5mm_P0.65mm_ClockwisePinNumbering", value: "BME280", at: [3, 0] });
const j1 = d.part("J1", { symbol: "Connector_Generic:Conn_01x04", footprint: "Connector_PinHeader_2.54mm:PinHeader_1x04_P2.54mm_Vertical", value: "I2C", at: [-7, 3.81] });
// pull-ups straight above the sensor's SDA and SCL pads, 3V3 at their top
const r2 = d.part("R2", { symbol: "Device:R", footprint: "Resistor_SMD:R_0603_1608Metric", value: "4k7", at: [3.325, 4.2], rotation: -90 });
const r1 = d.part("R1", { symbol: "Device:R", footprint: "Resistor_SMD:R_0603_1608Metric", value: "4k7", at: [5, 4.2], rotation: -90 });
const c1 = d.part("C1", { symbol: "Device:C", footprint: "Capacitor_SMD:C_0603_1608Metric", value: "100n", at: [6.5, 0], rotation: 90 });

d.net("3V3", j1.pin(1), u1.pin(8), u1.pin(6), u1.pin(2), r1.pin(1), r2.pin(1), c1.pin(1));
d.net("GND", j1.pin(2), u1.pin(1), u1.pin(7), u1.pin(5), c1.pin(2));
d.net("SDA", j1.pin(3), u1.pin(3), r2.pin(2));
d.net("SCL", j1.pin(4), u1.pin(4), r1.pin(2));
d.outline(rect(20, 16, { radius: 1.5 }));

// Tracks start and end on pins (`u1.pin(3)`), which stand for their pads' centres; the corners
// between are design coordinates (mm, y up from the board's centre). `files.pads` lists every pad.
const sig = { width: 0.2 }, pwr = { width: 0.25 };
// 3V3: a rail along the pull-ups' tops to the header, the sensor's three supply pads, C1
d.track("3V3", [j1.pin(1), [-5.5, 5.025], r1.pin(1)], pwr);              // passes R2.1 on the way
d.track("3V3", [u1.pin(2), [2.675, 2.55], [1.2, 2.55]], sig);           // clear of R2's pad (DRC: 0.075 mm when it ran straight up)
d.track("3V3", [[1.2, 5.025], [1.2, -2.8], [6.5, -2.8], c1.pin(1)], pwr);
d.track("3V3", [u1.pin(8), [2.025, -2.8]], sig);
d.track("3V3", [u1.pin(6), [3.325, -2.8]], sig);
// SDA and SCL: up to the pull-ups on top, then down through vias to the header on the bottom
d.track("SDA", [u1.pin(3), r2.pin(2)], sig);
d.via("SDA", [3.325, 2.6]);
d.track("SDA", [[3.325, 2.6], [0.3, 2.6], [0.3, -1.27], j1.pin(3)], { ...sig, layer: "B.Cu" });
d.track("SCL", [u1.pin(4), [3.975, 2.2], [5, 2.2], r1.pin(2)], sig);
d.via("SCL", [5, 2.2]);
d.track("SCL", [[5, 2.2], [5, -3.4], [-5.6, -3.4], j1.pin(4)], { ...sig, layer: "B.Cu" });
// GND: the bottom pour, and a via beside each ground pad on top
d.zone("GND", { layer: "B.Cu" });
for (const [pin, via] of [[u1.pin(1), [2.025, 1.9]], [u1.pin(7), [2.675, -1.9]], [u1.pin(5), [3.975, -1.9]], [c1.pin(2), [6.5, 1.9]]] as const) {
  d.track("GND", [pin, [via[0], via[1]]], sig);
  d.via("GND", [via[0], via[1]]);
}

const files = await d.write("out/bme280", { fill: true });
for (const n of files.notes) console.log("note:", n);
const at = (pin: string) => files.pads[pin]!.map((v) => v.toFixed(3)).join(", ");
console.log(`SDA runs from U1.3 (${at("U1.3")}) to J1.3 (${at("J1.3")})`);
const erc = await pcb.erc(files.schematic);
const drc = await pcb.drc(files.board, { parity: true });
for (const f of [...erc.findings, ...drc.findings]) console.log(`${f.severity}: ${f.rule}: ${f.message} ${f.refs.join(" ")}`);
console.log(`ERC ${erc.errors} error(s), DRC ${drc.errors} error(s)`);
const stepFile = await pcb.step(files.board);
await model("bme280_board", () => importStep(stepFile))();

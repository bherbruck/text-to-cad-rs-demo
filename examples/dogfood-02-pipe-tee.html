02-pipe-tee

A pipe tee: a run along X and a branch up Z, 32 mm OD / 24 mm bore, barbed ribs at both run ends (grip for a hose) and a 4-bolt flange on the branch.

#Source

Run it with cad run examples/dogfood/02-pipe-tee.ts.

// A pipe tee: a run along X and a branch up Z, 32 mm OD / 24 mm bore, barbed ribs at both run ends
// (grip for a hose) and a 4-bolt flange on the branch.
import { cylinder, model, param, torus } from "@cad/sdk";

export const tee = model("pipe_tee", () => {
  const od = param("od", 32, { min: 10, max: 80, unit: "mm" });
  const bore = param("bore", 24, { min: 4, max: 70, unit: "mm" });
  const run = 120, rise = 70;

  const main = cylinder(od / 2, run).rotateY(90);                                  // along X, centred
  const branch = cylinder(od / 2, rise, { align: ["center", "center", "min"] });   // up Z from the axis
  const flange = cylinder(od / 2 + 12, 6, { align: ["center", "center", "max"] }).translate(0, 0, rise);
  const boltHole = cylinder(2.75, 20).translate(od / 2 + 6, 0, rise - 3);
  // barbs: rings proud of the pipe near each run end
  const rib = torus(od / 2, 1.2).rotateY(90);
  const ribs = [-1, 1].flatMap((side) => [0, 1, 2].map((i) => rib.translate(side * (run / 2 - 6 - i * 7), 0, 0)));

  const solid = main.fuse(branch, flange, ribs);
  const holes = [cylinder(bore / 2, run + 2).rotateY(90), cylinder(bore / 2, rise + 2, { align: ["center", "center", "min"] })];
  return solid
    .cut(holes, boltHole.polarCopies(4, { degrees: 360 }))
    .label("tee", { color: "#c0c4c8", material: "pp" });
});

if (import.meta.main) await tee();
